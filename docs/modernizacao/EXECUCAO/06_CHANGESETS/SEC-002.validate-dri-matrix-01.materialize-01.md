## 3. Artefato — `scripts/sec002/validate-dri-matrix.mjs`

**Caminho de destino:** `scripts/sec002/validate-dri-matrix.mjs`
**Encoding:** UTF-8, LF, sem BOM. ESM puro. Node ≥ 20.

```js
#!/usr/bin/env node
// Validador determinístico da matriz DRI (D1–D10) do SEC-002.
// Consome scripts/sec002/load-apply-map.mjs. Nunca lê o apply-map JSON diretamente.
import { readFileSync, existsSync } from "node:fs";
import { loadApplyMap } from "./load-apply-map.mjs";

const NAME = "sec002/validate-dri-matrix";
const MATRIX_VERSION = "v2026-07-12.1";
const MAX_WINDOW_HOURS = 30;
const ROLES = [
  "dri_technical",
  "dri_platform",
  "dri_tests",
  "dri_observability",
  "dri_security",
  "comms_owner",
  "incident_commander",
  "gm_approver",
];

function fail(rule, applyId, detail) {
  console.error(`[${NAME}] FAIL - ${rule} em ${applyId}: ${detail}`);
}
function opError(msg) {
  console.error(`[${NAME}] FAIL - OP: ${msg}`);
  process.exit(2);
}

// --- CLI ---
const argv = process.argv.slice(2);
const args = {};
for (let i = 0; i < argv.length; i += 2) {
  const k = argv[i]; const v = argv[i + 1];
  if (!k || !v || !k.startsWith("--")) opError(`argumento inválido: ${k}`);
  args[k.slice(2)] = v;
}
if (!args["apply-id"] || !args["file"]) {
  opError("uso: --apply-id <id> --file <path>");
}
const applyId = args["apply-id"];
const filePath = args["file"];

// --- D1 ---
if (!existsSync(filePath)) opError(`arquivo ausente: ${filePath}`);
const raw = readFileSync(filePath, "utf8");
const fm = raw.match(/^---\r?\n([\s\S]*?)\r?\n---/);
if (!fm) { fail("D1", applyId, "frontmatter YAML ausente"); process.exit(1); }

// --- parser YAML mínimo (indent 2, escalares, mapas aninhados) ---
function parseYaml(src) {
  const lines = src.split(/\r?\n/);
  const root = {};
  const stack = [{ indent: -1, node: root }];
  for (const line of lines) {
    if (!line.trim() || /^\s*#/.test(line)) continue;
    const m = line.match(/^(\s*)([A-Za-z0-9_]+):\s*(.*)$/);
    if (!m) return null;
    const indent = m[1].length;
    const key = m[2];
    const val = m[3];
    while (stack.length > 1 && stack[stack.length - 1].indent >= indent) stack.pop();
    const parent = stack[stack.length - 1].node;
    if (val === "") {
      const child = {};
      parent[key] = child;
      stack.push({ indent, node: child });
    } else {
      let v = val.trim();
      if (v === "true") v = true;
      else if (v === "false") v = false;
      else if (v === "null" || v === "~") v = null;
      else v = v.replace(/^["'](.*)["']$/, "$1");
      parent[key] = v;
    }
  }
  return root;
}

const doc = parseYaml(fm[1]);
if (!doc) { fail("D1", applyId, "YAML inválido"); process.exit(1); }

const violations = [];
const V = (rule, detail) => { violations.push([rule, detail]); };

// --- D2 ---
let map;
try { map = loadApplyMap(); } catch (e) { opError(`loadApplyMap: ${e.message}`); }
const known = new Set(map.applies.map((a) => a.apply_id));
if (doc.sec002_apply_id !== applyId) V("D2", `sec002_apply_id=${doc.sec002_apply_id} != --apply-id=${applyId}`);
if (!known.has(applyId)) V("D2", `apply-id desconhecido no apply-map (${[...known].join(",")})`);

// --- D3 ---
if (doc.matrix_version !== MATRIX_VERSION) V("D3", `matrix_version=${doc.matrix_version} != ${MATRIX_VERSION}`);

// --- D4/D5/D6/D10 ---
const primaries = {};
for (const role of ROLES) {
  const r = doc.roles && doc.roles[role];
  if (!r || typeof r !== "object") { V("D4", `papel ausente: ${role}`); continue; }
  const { primary, backup, available } = r;
  if (!primary) V("D4", `${role}.primary ausente`);
  if (!backup) V("D4", `${role}.backup ausente`);
  if (available !== true && available !== false) V("D4", `${role}.available inválido`);
  if (primary && backup && primary === backup) V("D5", `${role}.primary == backup`);
  if (primary && /^<.*>$/.test(String(primary))) V("D10", `${role}.primary placeholder`);
  if (backup && /^<.*>$/.test(String(backup))) V("D10", `${role}.backup placeholder`);
  primaries[role] = primary;
}
if (primaries.gm_approver) {
  for (const role of ROLES) {
    if (role === "gm_approver") continue;
    if (primaries[role] && primaries[role] === primaries.gm_approver) {
      V("D6", `gm_approver.primary=${primaries.gm_approver} colide com ${role}`);
    }
  }
}

// --- D7 ---
if (!doc.approved_by) V("D7", "approved_by ausente");
else {
  if (doc.approved_by.dri_technical !== primaries.dri_technical) {
    V("D7", `approved_by.dri_technical (${doc.approved_by.dri_technical}) != roles.dri_technical.primary (${primaries.dri_technical})`);
  }
  if (doc.approved_by.gm !== primaries.gm_approver) {
    V("D7", `approved_by.gm (${doc.approved_by.gm}) != roles.gm_approver.primary (${primaries.gm_approver})`);
  }
}

// --- D8 ---
const s = doc.window && doc.window.starts_at;
const e = doc.window && doc.window.ends_at;
if (!s || !e) V("D8", "window.starts_at/ends_at ausente");
else {
  const ts = Date.parse(s); const te = Date.parse(e);
  if (Number.isNaN(ts) || Number.isNaN(te)) V("D8", "window ISO8601 inválido");
  else if (!(ts < te)) V("D8", "starts_at >= ends_at");
  else if ((te - ts) / 36e5 > MAX_WINDOW_HOURS) V("D8", `duração > ${MAX_WINDOW_HOURS}h`);
}

// --- D9 ---
const expect = {
  go_no_go: `.github/sec002-go-no-go/${applyId}.md`,
  comms: `.github/sec002-comms/${applyId}.md`,
  acceptance: `.github/sec002-acceptance/${applyId}.md`,
};
if (!doc.evidence_refs) V("D9", "evidence_refs ausente");
else {
  for (const k of Object.keys(expect)) {
    if (doc.evidence_refs[k] !== expect[k]) {
      V("D9", `evidence_refs.${k}=${doc.evidence_refs[k]} != ${expect[k]}`);
    }
  }
}

// --- D10 (varredura final de placeholders/campos vazios obrigatórios) ---
if (/<[^>\n]+>/.test(raw.slice(0, raw.indexOf("\n---", 4) + 4))) {
  V("D10", "placeholder <...> presente no frontmatter");
}
for (const [k, v] of Object.entries(doc.handoffs || {})) {
  if (k === "post_apply") continue;
  if (!v) V("D10", `handoffs.${k} vazio`);
}

if (violations.length === 0) {
  console.log(`[${NAME}] OK`);
  process.exit(0);
}
for (const [rule, detail] of violations) fail(rule, applyId, detail);
process.exit(1);
```

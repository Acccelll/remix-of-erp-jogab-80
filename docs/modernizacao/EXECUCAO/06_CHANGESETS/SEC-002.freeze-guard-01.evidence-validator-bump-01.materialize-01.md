## 2. Artefato 1 — `scripts/sec002/freeze-guard.mjs`

**Caminho:** `scripts/sec002/freeze-guard.mjs`
**Encoding:** UTF-8, LF, sem BOM. ESM puro. Node ≥ 20. Sem deps externas.

```js
#!/usr/bin/env node
import { execSync } from "node:child_process";
import { loadApplyMap, getApply } from "./load-apply-map.mjs";

const NAME = "sec002/freeze-guard";
const DRY_RUN = process.argv.includes("--dry-run");
const fail = [];
const push = (code, msg) => fail.push(`${code} ${msg}`);

const map = loadApplyMap();

const matchProtected = (file, pattern) => {
  if (pattern.endsWith("/**")) return file.startsWith(pattern.slice(0, -2));
  return file === pattern;
};

const PARSER_MAP = [
  { pattern: "scripts/sec002/parse-apply-id.mjs", allowed: ["parse-apply-id-01"], code: "E_FREEZE_F10" },
  { pattern: ".github/sec002-evidence/parse-apply-id-01/**", allowed: ["parse-apply-id-01"], code: "E_FREEZE_F11" }
];

const EVIDENCE_VALIDATOR_MAP = [
  { pattern: "scripts/sec002/validate-evidence-index.mjs", allowed: ["validate-evidence-index-01"], code: "E_FREEZE_F12" },
  { pattern: ".github/sec002-evidence/validate-evidence-index-01/**", allowed: ["validate-evidence-index-01"], code: "E_FREEZE_F13" },
  { pattern: ".github/sec002-evidence/INDEX.md", allowed: ["evidence-index-01", "validate-evidence-index-01"], code: "E_FREEZE_F14" }
];

const SPECIFIC_MAP = [...PARSER_MAP, ...EVIDENCE_VALIDATOR_MAP];

for (const entry of EVIDENCE_VALIDATOR_MAP) {
  if (!map.protected_paths.includes(entry.pattern)) {
    push("E_FREEZE_F9", `${entry.pattern} ausente de map.protected_paths[]`);
  }
}

let changed = [];
if (!DRY_RUN) {
  try {
    const out = execSync("git diff --name-only origin/main...HEAD", { encoding: "utf8" });
    changed = out.split("\n").map(s => s.trim()).filter(Boolean);
  } catch (e) {
    console.error(`[${NAME}] FAIL - não foi possível obter diff (${e.message})`);
    process.exit(1);
  }
}

const APPLY_ID = process.env.APPLY_ID || "";
let apply = null;
if (APPLY_ID) {
  try { apply = getApply(APPLY_ID); } catch { apply = null; }
}

const covers = (file) => {
  if (!apply) return false;
  return (apply.allowed_paths || []).some(p => matchProtected(file, p));
};

const specificRuleFor = (file) => SPECIFIC_MAP.find(entry => matchProtected(file, entry.pattern));

for (const file of changed) {
  const hits = map.protected_paths.filter(p => matchProtected(file, p));
  if (hits.length === 0) continue;

  const specific = specificRuleFor(file);
  const code = specific?.code || "E_FREEZE_F9";

  if (!APPLY_ID) { push(code, `${file} protegido mas PR sem APPLY_ID`); continue; }
  if (!apply) { push(code, `${file} protegido e APPLY_ID=${APPLY_ID} inválido`); continue; }

  if (specific && !specific.allowed.includes(APPLY_ID)) {
    push(code, `${file} só editável sob ${specific.allowed.join(" ou ")} (APPLY_ID=${APPLY_ID})`);
    continue;
  }

  if (!covers(file)) push(code, `${file} fora de allowed_paths[] de ${APPLY_ID}`);
}

if (fail.length === 0) { console.log(`[${NAME}] OK`); process.exit(0); }
console.error(`[${NAME}] FAIL`);
for (const v of fail) console.error(`  - ${v}`);
process.exit(1);
```

Contratos:

- `stdout` em sucesso: única linha `[sec002/freeze-guard] OK`; `exit 0`.
- `stderr` em falha: `[sec002/freeze-guard] FAIL` + uma violação por linha com código `E_FREEZE_F<n>`; `exit 1`.
- `--dry-run` não lê diff Git; valida apenas a consistência da tabela F12–F14 contra `map.protected_paths[]`.
- Nunca escreve em disco, nunca faz rede, nunca lê `.github/sec002-apply-map.json` direto — apenas via `loadApplyMap()`.
- Nunca reimplementa parsing de `APPLY_ID`; consome `process.env.APPLY_ID` populado pelo pré-gate `parse-apply-id`.

## 3. Artefato 2 — `.github/workflows/sec002-freeze-guard.yml`

**Caminho:** `.github/workflows/sec002-freeze-guard.yml`

```yaml
name: sec002-freeze-guard
on:
  pull_request:
    paths:
      - "**"

concurrency:
  group: sec002-freeze-guard-${{ github.ref }}
  cancel-in-progress: true

permissions:
  contents: read
  pull-requests: read

jobs:
  guard:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    env:
      PR_TITLE: ${{ github.event.pull_request.title }}
      PR_BODY: ${{ github.event.pull_request.body }}
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: actions/setup-node@v4
        with:
          node-version: "20"
      - name: parse-apply-id
        run: |
          APPLY_ID="$(node scripts/sec002/parse-apply-id.mjs)"
          echo "APPLY_ID=${APPLY_ID}" >> "$GITHUB_ENV"
      - name: validate apply-map
        run: node scripts/sec002/validate-apply-map.mjs
      - name: freeze-guard
        run: node scripts/sec002/freeze-guard.mjs
```

Contratos:

- Único pré-gate `parse-apply-id`; nenhum re-parse posterior.
- `fetch-depth: 0` para garantir que `git diff origin/main...HEAD` funcione.
- Sem segredos, sem write permissions, sem chamadas externas.
- Independente de `sec002-pr-checklist` — falha aqui bloqueia o PR mesmo que o checklist passe.

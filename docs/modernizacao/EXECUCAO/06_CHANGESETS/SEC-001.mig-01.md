## 3. Tabela temporária de plano de merge

```sql
CREATE TEMP TABLE _sec001_merge_plan ON COMMIT DROP AS
WITH candidatos AS (
  SELECT
    lower(split_part(email, '@', 1))              AS login,
    id,
    email,
    created_at,
    row_number() OVER (
      PARTITION BY lower(split_part(email, '@', 1))
      ORDER BY created_at ASC, id ASC
    ) AS rn,
    count(*)   OVER (PARTITION BY lower(split_part(email, '@', 1))) AS total
  FROM auth.users
  WHERE email LIKE '%@planifik.local'
     OR email LIKE '%@obraflow.local'
)
SELECT
  login,
  max(id) FILTER (WHERE rn = 1) AS keep_id,
  array_agg(id) FILTER (WHERE rn > 1) AS drop_ids,
  total
FROM candidatos
GROUP BY login, total;
```

## 4. Classe C2 — rename simples (`@planifik.local` → `@obraflow.local`)

```sql
-- Idempotente: só altera linhas que ainda estão no domínio legado
-- e cujo login canônico não existe também em @obraflow.local (evita colisão UNIQUE).
UPDATE auth.users u
SET email = lower(split_part(u.email, '@', 1)) || '@obraflow.local'
WHERE u.email LIKE '%@planifik.local'
  AND NOT EXISTS (
    SELECT 1 FROM auth.users u2
    WHERE u2.id <> u.id
      AND lower(split_part(u2.email, '@', 1)) = lower(split_part(u.email, '@', 1))
  );
```

## 5. Classe C3 — merge de duplicatas

Loop determinístico sobre `_sec001_merge_plan` onde `total > 1`:

```sql
DO $$
DECLARE
  r RECORD;
  drop_id uuid;
BEGIN
  FOR r IN
    SELECT login, keep_id, unnest(drop_ids) AS drop_id
    FROM _sec001_merge_plan
    WHERE total > 1
  LOOP
    drop_id := r.drop_id;

    -- 5.1 profiles: preservar is_gm/acessos via OR/união
    UPDATE public.profiles kp
    SET
      is_gm   = COALESCE(kp.is_gm,   false) OR COALESCE(dp.is_gm, false),
      acessos = COALESCE(kp.acessos, '{}'::jsonb) || COALESCE(dp.acessos, '{}'::jsonb),
      nome    = COALESCE(NULLIF(kp.nome, ''), dp.nome)
    FROM public.profiles dp
    WHERE kp.id = r.keep_id AND dp.id = drop_id;

    DELETE FROM public.profiles WHERE id = drop_id;

    -- 5.2 user_roles: reatribuir; unique (user_id, role) descarta duplicata
    INSERT INTO public.user_roles (user_id, role)
    SELECT r.keep_id, ur.role
    FROM public.user_roles ur
    WHERE ur.user_id = drop_id
    ON CONFLICT (user_id, role) DO NOTHING;

    DELETE FROM public.user_roles WHERE user_id = drop_id;

    -- 5.3 user_empresas: mesma lógica
    INSERT INTO public.user_empresas (user_id, empresa_id)
    SELECT r.keep_id, ue.empresa_id
    FROM public.user_empresas ue
    WHERE ue.user_id = drop_id
    ON CONFLICT (user_id, empresa_id) DO NOTHING;

    DELETE FROM public.user_empresas WHERE user_id = drop_id;

    -- 5.4 remover auth.users duplicado
    DELETE FROM auth.users WHERE id = drop_id;
  END LOOP;
END $$;
```

## 8. Idempotência

- §4 usa `NOT EXISTS` — segunda execução é no-op.
- §5 opera apenas sobre `total > 1`; após a primeira execução `_sec001_merge_plan` fica vazio na classe C3.
- §6 é `UPDATE ... WHERE LIKE '%@planifik.local'` — no-op após §4/§6.
- §7 são `SELECT` puros.

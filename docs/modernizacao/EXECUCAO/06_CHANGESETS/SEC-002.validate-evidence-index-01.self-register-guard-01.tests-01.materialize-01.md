```js
#!/usr/bin/env node
// Runner determinístico T1–T5 para o self-register-guard do validate-evidence-index.
// Não escreve no INDEX.md real; cria fixtures descartáveis em tmp/.

import { spawnSync } from "node:child_process";
import {
  cpSync,
  existsSync,
  mkdirSync,
  readFileSync,
  rmSync,
  writeFileSync,
} from "node:fs";
import { dirname, join, resolve } from "node:path";
import { fileURLToPath } from "node:url";

const NAME = "sec002/test-validate-evidence-index-self-register-guard";
const __filename = fileURLToPath(import.meta.url);
const PROJECT_ROOT = resolve(dirname(__filename), "../..");
const VALIDATOR = join(PROJECT_ROOT, "scripts/sec002/validate-evidence-index.mjs");
const CANONICAL_INDEX = join(PROJECT_ROOT, ".github/sec002-evidence/INDEX.md");
const TMP_ROOT = join(PROJECT_ROOT, "tmp/sec002-validate-evidence-index-self-register-guard");
const FIXTURE_INDEX = ".github/sec002-evidence/INDEX.md";
const DUPLICATE_APPLY_ID = "evidence-index-evidence-validator-bump-01";

function fail(testId, detail, exitCode = 1) {
  console.error(`[${NAME}] FAIL ${testId}: ${detail}`);
  process.exit(exitCode);
}

function ok(testId) {
  console.log(`[${NAME}] OK ${testId}`);
}

function ensure(condition, testId, detail) {
  if (!condition) fail(testId, detail);
}

function splitRow(line) {
  if (!line.startsWith("|") || !line.endsWith("|")) return null;
  return line.slice(1, -1).split("|").map((cell) => cell.trim());
}

function parseIndex(raw) {
  const lines = raw.split(/\r?\n/);
  const header = "| # | apply_id | manifest | changeset docs | schema | refs |";
  const headerIndex = lines.findIndex((line) => line.trim() === header);
  if (headerIndex < 0) fail("OP", "tabela canônica ausente", 2);

  const rows = [];
  for (let i = headerIndex + 2; i < lines.length; i += 1) {
    const line = lines[i].trim();
    if (!line || !line.startsWith("|")) break;
    const cells = splitRow(line);
    if (!cells || cells.length !== 6) fail("OP", `linha inválida no índice canônico: ${i + 1}`, 2);
    rows.push({ lineNumber: i + 1, raw: lines[i], cells });
  }

  const schemaLine = lines.find((line) => line.startsWith("Apply-map:"));
  const schemaVersion = schemaLine?.replace("Apply-map:", "").trim();
  if (!schemaVersion) fail("OP", "Apply-map ausente no índice canônico", 2);

  return { lines, rows, schemaVersion };
}

function writeFixtureFile(root, relativePath, content) {
  const target = join(root, relativePath);
  mkdirSync(dirname(target), { recursive: true });
  writeFileSync(target, content, "utf8");
}

function copyIfExistsIntoFixture(root, relativePath, fallbackContent) {
  const source = join(PROJECT_ROOT, relativePath);
  const target = join(root, relativePath);
  mkdirSync(dirname(target), { recursive: true });
  if (existsSync(source)) cpSync(source, target);
  else writeFileSync(target, fallbackContent, "utf8");
}

function prepareBaseFixture(caseId) {
  if (!existsSync(VALIDATOR)) fail("OP", `validador ausente: ${VALIDATOR}`, 2);
  if (!existsSync(CANONICAL_INDEX)) fail("OP", `INDEX.md canônico ausente: ${CANONICAL_INDEX}`, 2);

  const raw = readFileSync(CANONICAL_INDEX, "utf8");
  const parsed = parseIndex(raw);
  if (parsed.rows.length !== 16) {
    fail("OP", `estado pós-R71 esperado com 16 linhas; encontrado ${parsed.rows.length}`, 2);
  }

  const root = join(TMP_ROOT, caseId);
  rmSync(root, { recursive: true, force: true });
  mkdirSync(root, { recursive: true });
  writeFixtureFile(root, FIXTURE_INDEX, raw);

  for (const row of parsed.rows) {
    const [, applyId, manifestPath, changesetPath, schema, refs] = row.cells;
    const manifestSource = join(PROJECT_ROOT, manifestPath);
    const manifestFallback = `${JSON.stringify({
      sec002_apply_id: applyId,
      schema_version: schema,
      source: changesetPath,
      changed_files: [],
      checks: [],
      hashes: {},
      refs: refs.split(",").map((ref) => ref.trim()).filter(Boolean),
    }, null, 2)}\n`;
    copyIfExistsIntoFixture(root, manifestPath, manifestFallback);
    copyIfExistsIntoFixture(root, changesetPath, `# fixture ${applyId}\n`);
  }

  return { root, parsed };
}

function runValidator(root) {
  const result = spawnSync(process.execPath, [VALIDATOR, "--file", FIXTURE_INDEX], {
    cwd: root,
    encoding: "utf8",
    shell: false,
  });
  return {
    status: result.status ?? 2,
    stdout: result.stdout || "",
    stderr: result.stderr || "",
    output: `${result.stdout || ""}\n${result.stderr || ""}`,
  };
}

function mutateIndex(root, mutate) {
  const target = join(root, FIXTURE_INDEX);
  const current = readFileSync(target, "utf8");
  writeFileSync(target, mutate(current), "utf8");
}

function appendLineBeforeTableEnd(raw, line) {
  const lines = raw.split(/\r?\n/);
  let insertAt = -1;
  for (let i = 0; i < lines.length; i += 1) {
    if (lines[i].startsWith("| 16 |")) insertAt = i + 1;
  }
  if (insertAt < 0) fail("OP", "linha 16 ausente para append de fixture", 2);
  lines.splice(insertAt, 0, line);
  return lines.join("\n");
}

function runCase(caseId, arrange, assertResult) {
  let root;
  try {
    const fixture = prepareBaseFixture(caseId);
    root = fixture.root;
    arrange?.(fixture);
    const result = runValidator(root);
    assertResult(result, fixture);
    ok(caseId);
  } finally {
    if (root?.startsWith(TMP_ROOT)) rmSync(root, { recursive: true, force: true });
  }
}

rmSync(TMP_ROOT, { recursive: true, force: true });

runCase("T1", null, (result) => {
  ensure(result.status === 0, "T1", `exit ${result.status}; output=${result.output}`);
  ensure(result.output.includes("[sec002/validate-evidence-index] OK"), "T1", "OK ausente");
});

runCase("T2", ({ root, parsed }) => {
  writeFixtureFile(
    root,
    ".github/sec002-evidence/nova-evid/manifest.json",
    `${JSON.stringify({
      sec002_apply_id: "nova-evid",
      schema_version: parsed.schemaVersion,
      source: "docs/modernizacao/EXECUCAO/06_CHANGESETS/SEC-002.fixture-R75.md",
      changed_files: [],
      checks: [],
      hashes: {},
      refs: ["R75"],
    }, null, 2)}\n`,
  );
}, (result) => {
  ensure(result.status === 1, "T2", `exit esperado 1; recebido ${result.status}; output=${result.output}`);
  ensure(result.output.includes("E_EVIDX_S1"), "T2", "E_EVIDX_S1 ausente");
  ensure(result.output.includes("apply_id=nova-evid"), "T2", "apply_id=nova-evid ausente");
});

runCase("T3", ({ root, parsed }) => {
  const row16 = parsed.rows.find((row) => row.cells[1] === DUPLICATE_APPLY_ID);
  if (!row16) fail("OP", `linha canônica ausente: ${DUPLICATE_APPLY_ID}`, 2);
  mutateIndex(root, (raw) => appendLineBeforeTableEnd(raw, row16.raw));
}, (result) => {
  ensure(result.status === 1, "T3", `exit esperado 1; recebido ${result.status}; output=${result.output}`);
  const acceptsS2 = result.output.includes("E_EVIDX_S2")
    && result.output.includes(`apply_id=${DUPLICATE_APPLY_ID}`)
    && result.output.includes("occurrences=2");
  const acceptsE3 = result.output.includes("E_EVIDX_E3");
  ensure(acceptsS2 || acceptsE3, "T3", `nem E_EVIDX_S2 nem E_EVIDX_E3 encontrados; output=${result.output}`);
});

runCase("T4", ({ root }) => {
  mkdirSync(join(root, ".github/sec002-evidence/tmp"), { recursive: true });
}, (result) => {
  ensure(result.status === 0, "T4", `exit ${result.status}; output=${result.output}`);
  ensure(result.output.includes("[sec002/validate-evidence-index] OK"), "T4", "OK ausente");
  ensure(!result.output.includes("E_EVIDX_S1"), "T4", "E_EVIDX_S1 indevido para diretório sem manifest");
});

runCase("T5", ({ root, parsed }) => {
  writeFixtureFile(root, "docs/modernizacao/EXECUCAO/06_CHANGESETS/SEC-002.fixture-R75.md", "# fixture R75\n");
  const line = `| 17 | missing-manifest-t5 | .github/sec002-evidence/missing-manifest-t5/manifest.json | docs/modernizacao/EXECUCAO/06_CHANGESETS/SEC-002.fixture-R75.md | ${parsed.schemaVersion} | R75 |`;
  mutateIndex(root, (raw) => appendLineBeforeTableEnd(raw, line));
}, (result) => {
  ensure(result.status === 1, "T5", `exit esperado 1; recebido ${result.status}; output=${result.output}`);
  ensure(result.output.includes("E_EVIDX_E4"), "T5", "E_EVIDX_E4 ausente");
  ensure(!result.output.includes("E_EVIDX_S1"), "T5", "E_EVIDX_S1 indevido após falha E4");
  ensure(!result.output.includes("E_EVIDX_S2"), "T5", "E_EVIDX_S2 indevido após falha E4");
});

rmSync(TMP_ROOT, { recursive: true, force: true });
console.log(`[${NAME}] OK`);
```

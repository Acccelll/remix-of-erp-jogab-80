```yaml
# sec002-pr-checklist
# apply-map: v2026-07-12.3
# consumers: parse-apply-id (parser) + 11 gates
name: sec002-pr-checklist
on:
  pull_request:
    paths:
      - "**"

concurrency:
  group: sec002-pr-checklist-${{ github.ref }}
  cancel-in-progress: true

permissions:
  contents: read
  pull-requests: read

jobs:
  checklist:
    runs-on: ubuntu-latest
    timeout-minutes: 8
    env:
      PR_TITLE: ${{ github.event.pull_request.title }}
      PR_BODY: ${{ github.event.pull_request.body }}
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: actions/setup-node@v4
        with:
          node-version: "20"

      - name: parse-apply-id  # consumer id="parse-apply-id" category="parser"
        run: |
          APPLY_ID="$(node scripts/sec002/parse-apply-id.mjs)"
          echo "APPLY_ID=${APPLY_ID}" >> "$GITHUB_ENV"

      - name: validate-apply-map
        run: node scripts/sec002/validate-apply-map.mjs

      - name: validate-pr-checklist
        run: node scripts/sec002/validate-pr-checklist.mjs

      - name: validate-changed-files
        run: node scripts/sec002/validate-changed-files.mjs

      - name: validate-staging-parity
        run: node scripts/sec002/validate-staging-parity.mjs

      - name: validate-go-no-go
        run: node scripts/sec002/validate-go-no-go.mjs

      - name: validate-comms-log
        run: node scripts/sec002/validate-comms-log.mjs

      - name: validate-dri-matrix
        run: node scripts/sec002/validate-dri-matrix.mjs

      - name: validate-acceptance-gates
        run: node scripts/sec002/validate-acceptance-gates.mjs

      - name: validate-evidence-archive
        run: node scripts/sec002/validate-evidence-archive.mjs

      - name: sec002-consumers-guard
        run: node scripts/sec002/check-consumers.mjs

      - name: sec002-freeze-guard
        run: node scripts/sec002/freeze-guard.mjs

      - name: sec-002-tests
        run: node scripts/sec002/run-tests.mjs

      - name: summary
        if: success()
        run: echo "gates: 11/11 OK (apply-map v2026-07-12.3, APPLY_ID=${APPLY_ID})"
```

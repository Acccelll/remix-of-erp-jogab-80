```js
// --- self-register-guard (R72/R73) ---------------------------------------
// Executa após E1–E8. Se algum E1–E8 falhou, o processo já terminou (S5).
import { readdirSync, statSync, existsSync } from 'node:fs';
import { join } from 'node:path';

const EVID_ROOT = '.github/sec002-evidence';

function enumerateApplyIdsOnDisk(root) {
  if (!existsSync(root)) return [];
  const out = [];
  for (const name of readdirSync(root)) {
    const dir = join(root, name);
    let st;
    try { st = statSync(dir); } catch { continue; }
    if (!st.isDirectory()) continue;
    const manifest = join(dir, 'manifest.json');
    if (!existsSync(manifest)) continue; // S4: diretório sem manifest.json é ignorado
    out.push(name);
  }
  return out.sort();
}

// `indexRows` é a lista de linhas 11..N já parseadas por E1–E8 acima.
// Cada elemento tem { lineNumber, applyId, manifestPath, changesetPath, schema, refs }.
// (Este contrato é o mesmo consumido por E3/E4/E5 em R61.)
const indexedApplyIds = indexRows.map(r => r.applyId);

// S2 reforço: duplicatas em INDEX.md
const seen = new Map();
for (const id of indexedApplyIds) {
  seen.set(id, (seen.get(id) ?? 0) + 1);
}
const s2Errors = [];
for (const [id, n] of seen) {
  if (n > 1) {
    s2Errors.push(
      `E_EVIDX_S2 apply_id=${id} occurrences=${n} ` +
      `hint="cada apply_id deve aparecer exatamente uma vez em INDEX.md"`
    );
  }
}

// S1: manifest físico sem linha correspondente
const filesOnDisk = enumerateApplyIdsOnDisk(EVID_ROOT);
const indexedSet = new Set(indexedApplyIds);
const s1Errors = [];
for (const id of filesOnDisk) {
  if (!indexedSet.has(id)) {
    s1Errors.push(
      `E_EVIDX_S1 apply_id=${id} manifest=${EVID_ROOT}/${id}/manifest.json ` +
      `hint="adicione linha no INDEX.md apontando para este manifest"`
    );
  }
}

if (s1Errors.length || s2Errors.length) {
  for (const e of s2Errors) console.error(e);
  for (const e of s1Errors) console.error(e);
  process.exit(1);
}
// --- fim self-register-guard --------------------------------------------

console.log('[sec002/validate-evidence-index] OK');
```

- **Mensagem de sucesso inalterada** — `[sec002/validate-evidence-index] OK` (compat. com o gate final do workflow em R67).
- **Ordem de erros** — `E_EVIDX_S2` antes de `E_EVIDX_S1` para tornar duplicatas visíveis primeiro; ambos anteriores a `exit(1)`.
